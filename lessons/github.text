
git init — инициализация локального репозитория.

git remote add origin <ссылка> — привязка удаленного репозитория к локальному.

git status — проверка статуса файлов в репозитории.

git add . — добавление всех измененных файлов в буфер обмена (staging area).

git commit -m "сообщение" — создание коммита (сохранение изменений из буфера).

git push origin main (или master) — отправка коммитов в удаленный репозиторий на GitHub.

git restore . (или путь к файлу) — откат изменений в рабочей зоне.

git reset . (или путь к файлу) — отмена git add (вывод файлов из буфера обмена назад).

git branch name_branch — создание новой ветки.

git switch name_branch (или git checkout name_branch) — переключение на ветку.

git switch -c name_branch (или git switch -b) — создание ветки и мгновенное переключение на неё.

git reset --mixed HEAD~1 — удаление последнего коммита с сохранением измененных файлов в папке.
